cmake_minimum_required(VERSION 3.20)

project(GravitySimulation)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_executable(GravitySimulator
    src/main.cpp
    src/glad.c

    src/engine/Gravity.cpp
    src/engine/Integrator.cpp
    src/engine/OctreeNode.cpp
    src/engine/Collisions.cpp
    src/engine/Physics.cpp

    src/graphics/Window.cpp
    src/graphics/Shader.cpp

    src/config/Config.cpp

)

target_include_directories(GravitySimulator PRIVATE
    include
)

include(FetchContent)

FetchContent_Declare(
    json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.12.0
)

FetchContent_Declare(
    glfw
    GIT_REPOSITORY https://github.com/glfw/glfw.git
    GIT_TAG 3.4
)

FetchContent_Declare(
    glm
    GIT_REPOSITORY https://github.com/g-truc/glm.git
    GIT_TAG 1.0.1
)

FetchContent_MakeAvailable(
    json
    glfw
    glm
)

find_package(OpenGL REQUIRED)
find_package(OpenMP)

target_link_libraries(GravitySimulator PRIVATE
    nlohmann_json::nlohmann_json
    glfw
    OpenGL::GL
)

if(OpenMP_CXX_FOUND)
    target_link_libraries(GravitySimulator PRIVATE OpenMP::OpenMP_CXX)
endif()

if(MSVC)
    target_compile_options(GravitySimulator PRIVATE
        $<$<CONFIG:Debug>:/Od>
        $<$<CONFIG:Release>:/O2>
    )
else()
    target_compile_options(GravitySimulator PRIVATE
        $<$<CONFIG:Debug>:-O0 -g>
        $<$<CONFIG:Release>:-O3 -march=native>
        $<$<CONFIG:RelWithDebInfo>:-O2 -g>
    )
endif()

set_target_properties(GravitySimulator PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
)
