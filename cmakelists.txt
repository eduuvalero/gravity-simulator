cmake_minimum_required(VERSION 3.20)

project(GravitySimulation)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_executable(GravitySimulator
    src/main.cpp

    src/engine/Gravity.cpp
    src/engine/Integrator.cpp
    src/engine/OctreeNode.cpp
    src/engine/Physics.cpp

    src/config/Config.cpp
    include/config/config.h
)

target_include_directories(GravitySimulator PRIVATE
    include
)

include(FetchContent)

FetchContent_Declare(
    json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.12.0
)

FetchContent_MakeAvailable(json)

target_link_libraries(GravitySimulator PRIVATE
    nlohmann_json::nlohmann_json
)
